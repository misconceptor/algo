#FLASHING DRIVE:
lsblk
if auto mounted : sudo umount /dev/sda*

sudo dd if=(file.iso) of=/dev/sdb bs=4M status=progress oflag=sync

sync (write it to flush buffers)

sudo umount /dev/(path)
sudo eject /dev/(path)
